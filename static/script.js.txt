document.getElementById('btn-enviar').addEventListener('click', enviarMensaje);

async function enviarMensaje() {
    const inputTexto = document.getElementById('mensaje-input');
    const inputArchivo = document.getElementById('archivo-input');
    const contenedorMensajes = document.getElementById('mensajes');

    const texto = inputTexto.value.trim();
    const archivo = inputArchivo.files[0];

    if (!texto && !archivo) return;

    // Mostrar mensaje del usuario en pantalla
    contenedorMensajes.innerHTML += `<div><b>Tú:</b> ${texto} ${archivo ? '[Archivo adjunto]' : ''}</div>`;

    // Preparar los datos para enviarlos por HTTP (soporta texto y archivos a la vez)
    const formData = new FormData();
    formData.append('mensaje', texto);
    if (archivo) {
        formData.append('archivo', archivo);
    }

    inputTexto.value = '';
    inputArchivo.value = '';

    try {
        const respuesta = await fetch('/chat', {
            method: 'POST',
            body: formData // Usamos FormData para enviar archivos y texto juntos
        });
        
        const datos = await respuesta.json();
        contenedorMensajes.innerHTML += `<div><b>IA:</b> ${datos.respuesta}</div>`;
        contenedorMensajes.scrollTop = contenedorMensajes.scrollHeight;
    } catch (error) {
        contenedorMensajes.innerHTML += `<div><b>Error:</b> No se pudo conectar con el servidor.</div>`;
    }
}
